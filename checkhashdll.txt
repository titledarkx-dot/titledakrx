#ifndef INTEGRITY_CHECK_DLL_H
#define INTEGRITY_CHECK_DLL_H

#include <windows.h>
#include <wininet.h>
#include <wincrypt.h>
#include <string>
#include <sstream>
#include <iomanip>

#pragma comment(lib, "wininet.lib")

namespace Security {

  
    inline std::string FetchRemoteHash(const std::string& rawUrl) {
        std::string response = "";
        HINTERNET hInternet = InternetOpenA("DLLIntegrityAgent", INTERNET_OPEN_TYPE_DIRECT, NULL, NULL, 0);
        if (!hInternet) return "";

        HINTERNET hConnect = InternetOpenUrlA(hInternet, rawUrl.c_str(), NULL, 0, INTERNET_FLAG_RELOAD | INTERNET_FLAG_SECURE, 0);
        if (hConnect) {
            char buffer[256];
            DWORD bytesRead = 0;
            while (InternetReadFile(hConnect, buffer, sizeof(buffer) - 1, &bytesRead) && bytesRead > 0) {
                buffer[bytesRead] = '\0';
                response += buffer;
            }
            InternetCloseHandle(hConnect);
        }
        InternetCloseHandle(hInternet);

        size_t lastPos = response.find_last_not_of(" \n\r\t");
        if (lastPos != std::string::npos) {
            response.erase(lastPos + 1);
        }
        return response;
    }


    inline std::string CalculateDLLSHA256(HMODULE hDllModule) {
        WCHAR szPath[MAX_PATH];
      
        if (GetModuleFileNameW(hDllModule, szPath, MAX_PATH) == 0) return "";

        HANDLE hFile = CreateFileW(szPath, GENERIC_READ, FILE_SHARE_READ, NULL, OPEN_EXISTING, FILE_ATTRIBUTE_NORMAL, NULL);
        if (hFile == INVALID_HANDLE_VALUE) return "";

        HCRYPTPROV hProv = 0;
        HCRYPTHASH hHash = 0;
        BYTE buffer[4096];
        DWORD bytesRead = 0;
        std::string hashString = "";

        if (CryptAcquireContext(&hProv, NULL, NULL, PROV_RSA_AES, CRYPT_VERIFYCONTEXT)) {
            if (CryptCreateHash(hProv, CALG_SHA_256, 0, 0, &hHash)) {
                while (ReadFile(hFile, buffer, sizeof(buffer), &bytesRead, NULL) && bytesRead > 0) {
                    CryptHashData(hHash, buffer, bytesRead, 0);
                }

                BYTE hashValue[32];
                DWORD hashLen = 32;
                if (CryptGetHashParam(hHash, HP_HASHVAL, hashValue, &hashLen, 0)) {
                    std::stringstream ss;
                    for (DWORD i = 0; i < hashLen; i++) {
                        ss << std::hex << std::setw(2) << std::setfill('0') << (int)hashValue[i];
                    }
                    hashString = ss.str();
                }
                CryptDestroyHash(hHash);
            }
            CryptReleaseContext(hProv, 0);
        }
        CloseHandle(hFile);
        return hashString;
    }


    inline bool VerifyDllIntegrity(HMODULE hDllModule, const std::string& githubRawUrl) {
        std::string expectedHash = FetchRemoteHash(githubRawUrl);
        if (expectedHash.empty()) {
            return false;
        }

        std::string localHash = CalculateDLLSHA256(hDllModule);
        if (localHash.empty()) {
            return false;
        }

        return (_stricmp(expectedHash.c_str(), localHash.c_str()) == 0);
    }
}
